cmake_minimum_required(VERSION 3.18)

set(SOURCES main.cpp OptionParser.cpp OptionParser.h)

if(CMAKE_SYSTEM MATCHES Windows)
    message("Building Windows executable")
    add_executable(cv710userspace WIN32 ${SOURCES} appicon.rc)
    target_include_directories(cv710userspace PRIVATE liblgx)
    target_link_libraries(cv710userspace PRIVATE mingw32 liblgx)
    target_link_options(cv710userspace PRIVATE -static)
else()
    find_library(SYSTEMD_LIB systemd)
    find_library(ASYNCNS_LIB asyncns)

    add_executable(cv710userspace ${SOURCES})
    target_include_directories(cv710userspace PRIVATE liblgx)
    target_link_libraries(cv710userspace PRIVATE liblgx)
    if(SYSTEMD_LIB)
        target_link_libraries(cv710userspace PRIVATE ${SYSTEMD_LIB})
    endif()
    if(ASYNCNS_LIB)
        target_link_libraries(cv710userspace PRIVATE ${ASYNCNS_LIB})
    endif()
endif()
